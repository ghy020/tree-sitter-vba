Attribute VB_Name = "RealWorld"
Option Explicit

Public Function GetObjectOrNothing() As Object
    If GetObjectOrNothing Is Nothing Then Exit Function
End Function

Public Sub Run(Optional ByVal showMessage As Boolean = True)
    Dim r As Long
    If r < 2 Then r = 2
    Err.Raise vbObjectError + 2000, , _
        "failure"
    MsgBox "done", vbInformation, "title"
    WalkControl session.Children.ElementAt(0), r, showMessage
End Sub
